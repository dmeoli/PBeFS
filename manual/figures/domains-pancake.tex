% a pancake of size #3 (0..5) centered at x = #1, row #2 (0 = top), style #4
\newcommand{\pancake}[4]{%
  \fill[#4, rounded corners=2pt] ($(#1,-#2*0.62)+(-0.32-0.27*#3,-0.24)$)
    rectangle ($(#1,-#2*0.62)+(0.32+0.27*#3,0.24)$);
  \node[font=\sffamily\small] at (#1,-#2*0.62) {#3};}
% a gap marker between the rows #2 and #2+1 of the stack at x = #1
\newcommand{\gapmark}[2]{%
  \fill[pbefsAdded] ($(#1,-#2*0.62-0.31)+(2.45,0)$) circle (1.6mm);}
\begin{document}
\begin{tikzpicture}
% the stack p = (3, 1, 2, 0, 4, 5), the top 4 pancakes about to be flipped
\node[steplabel] at (0,1.0) {$p$, $h = 3$};
\pancake{0}{0}{3}{fill=pbefsPrimary!55}
\pancake{0}{1}{1}{fill=pbefsPrimary!55}
\pancake{0}{2}{2}{fill=pbefsPrimary!55}
\pancake{0}{3}{0}{fill=pbefsPrimary!55}
\pancake{0}{4}{4}{fill=pbefsTint}
\pancake{0}{5}{5}{fill=pbefsTint}
\fill[pbefsInk!45] (-2.3,-3.72) rectangle (2.3,-3.86);
\node[val, anchor=west] at (2.45,-4.1) {plate};
\gapmark{0}{0}
\gapmark{0}{2}
\gapmark{0}{3}
\draw[->, line width=1.1pt, draw=pbefsInk!45] (3.0,-1.55) -- (4.8,-1.55)
  node[midway, above, val] {flip of 4};
% the stack after the flip, (0, 2, 1, 3, 4, 5)
\begin{scope}[xshift=7.8cm]
\node[steplabel] at (0,1.0) {$t$, $h = 2$};
\pancake{0}{0}{0}{fill=pbefsAdded!55}
\pancake{0}{1}{2}{fill=pbefsAdded!55}
\pancake{0}{2}{1}{fill=pbefsAdded!55}
\pancake{0}{3}{3}{fill=pbefsAdded!55}
\pancake{0}{4}{4}{fill=pbefsTint}
\pancake{0}{5}{5}{fill=pbefsTint}
\fill[pbefsInk!45] (-2.3,-3.72) rectangle (2.3,-3.86);
\node[val, anchor=west] at (2.45,-4.1) {plate};
\gapmark{0}{0}
\gapmark{0}{2}
\end{scope}
\node[caption, anchor=west] at (-2.4,-4.75)
  {dots: the gaps; only the pair at the boundary of the flip changes,
   from the gap $(0, 4)$ to the pair $(3, 4)$};
\end{tikzpicture}
