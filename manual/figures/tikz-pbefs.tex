% tikz-pbefs.tex: the TikZ styles shared by the figures of the manual, \input
% by each standalone figure after \usepackage{lamalunga}. The grammar is
% that of flat diagrams (filled shapes, one color per role, no outlines,
% gradients or shadows); the colors are computed by lamalunga from one
% primary, so that changing \pbefsprimary recolours every figure:
%   cell, touched, added, free    the places of a vector: in use, read or
%                                 followed by the operation, added by it,
%                                 free;
%   node, vertex                  a node of a linked structure (box) and a
%                                 vertex of a tree or graph (disc), each
%                                 with -touched and -added variants;
%   edge, edge-touched            the edges of a tree or graph, and those
%                                 an operation follows;
%   ptr, var, index               a pointer, a variable, an index.
% \cells{name}{x,y}{list} draws a row of places (items text/style),
% \indices{name}{n} the indices 0 .. n-1 above them.
\usetikzlibrary{arrows.meta,positioning,calc,fit,shapes.multipart,matrix,
                backgrounds,shapes.geometric,decorations.pathreplacing}
\providecommand{\pbefsprimary}{7030A0}
\definecolor{pbefsPrimary}{HTML}{\pbefsprimary}
\colorharmony{pbefs}{pbefsPrimary}
\colorlet{pbefsAdded}{pbefs-triad-a}
\colorlet{pbefsInk}{black!78}
\colorlet{pbefsTint}{\goldentint{pbefsPrimary}{4}}
\colorlet{pbefsFree}{\goldentint{pbefsPrimary}{7}}
\readableon{pbefsOnPrimary}{pbefsPrimary}{black}{white}
\readableon{pbefsOnAdded}{pbefsAdded}{black}{white}
\tikzset{
  pbefs/.style={>={Stealth[length=2.2mm,width=2mm]},font=\small,
              every node/.append style={inner sep=0pt},line cap=round},
  cell/.style={rounded corners=1.6pt,minimum width=7mm,minimum height=7mm,
               font=\small\ttfamily,fill=pbefsTint,text=pbefsInk,
               draw=white,line width=1.2pt,outer sep=0pt},
  touched/.style={cell,fill=pbefsPrimary,text=pbefsOnPrimary,
                  font=\small\ttfamily\bfseries},
  added/.style={cell,fill=pbefsAdded,text=pbefsOnAdded,
                font=\small\ttfamily\bfseries},
  free/.style={cell,fill=pbefsFree,text=pbefsInk!55},
  node/.style={rounded corners=2pt,minimum width=8mm,minimum height=6mm,
               font=\small\ttfamily,fill=pbefsTint,text=pbefsInk},
  node-touched/.style={node,fill=pbefsPrimary,text=pbefsOnPrimary,
                       font=\small\ttfamily\bfseries},
  node-added/.style={node,fill=pbefsAdded,text=pbefsOnAdded,
                     font=\small\ttfamily\bfseries},
  vertex/.style={circle,minimum size=7.5mm,font=\small\ttfamily,
                 fill=pbefsTint,text=pbefsInk},
  vertex-touched/.style={vertex,fill=pbefsPrimary,text=pbefsOnPrimary,
                         font=\small\ttfamily\bfseries},
  vertex-added/.style={vertex,fill=pbefsAdded,text=pbefsOnAdded,
                       font=\small\ttfamily\bfseries},
  edge/.style={line width=1.1pt,draw=pbefsInk!45},
  edge-touched/.style={line width=1.8pt,draw=pbefsPrimary},
  ptr/.style={->,line width=1.1pt,draw=pbefsInk},
  var/.style={font=\small\itshape,text=pbefsPrimary},
  index/.style={font=\scriptsize\ttfamily,text=pbefsInk!55},
  nullptr/.style={font=\small,text=pbefsInk!55}
}
\newcommand{\cells}[3]{%
  \foreach \v/\s [count=\i from 0] in {#3}
    \node[\s] (#1-\i) at ($(#2)+(\i*7.4mm,0)$) {\v};}
\newcommand{\indices}[2]{%
  \foreach \i in {0,...,\numexpr#2-1\relax}
    \node[index,above=1.6mm of #1-\i] {\i};}
% --- the roles of the figures of a search, in the same flat grammar -------
%   open, closed, unseen         a state generated and in OPEN (light
%                                tint), expanded (darker tint), not
%                                generated yet (lightest);
%   current                      the state being expanded (primary);
%   goal, reopened, pruned       a goal, a state reopened or generated by
%                                the step (added), a state pruned;
%   owner=k, ownerbox=k          the state or the box of the worker k
%                                (k = 0..3): the primary and the triad
%                                color at two tints each;
%   path, msg                    the arcs of an optimal path, a message;
%   cellhi, cellnew              aliases of touched and added;
%   worker                       the background box of a worker;
%   val, cost, listname, caption, steplabel   the texts.
% The colors pb... of the first figures are kept as aliases of the roles.
% the goal and the incumbent: the primary darkened, the four owners: the
% primary and the triad color, each at two golden tints; no other hue
\colorlet{pbefsGoal}{\goldenshade{pbefsPrimary}{1}}
\readableon{pbefsOnGoal}{pbefsGoal}{black}{white}
\colorlet{pbefsClosed}{\goldentint{pbefsPrimary}{2}}
\readableon{pbefsOnClosed}{pbefsClosed}{black}{white}
\colorlet{pbefsOwner0}{\goldentint{pbefsPrimary}{1}}
\colorlet{pbefsOwner1}{\goldentint{pbefsAdded}{1}}
\colorlet{pbefsOwner2}{\goldentint{pbefsPrimary}{3}}
\colorlet{pbefsOwner3}{\goldentint{pbefsAdded}{3}}
\readableon{pbefsOnOwner0}{pbefsOwner0}{black}{white}
\readableon{pbefsOnOwner1}{pbefsOwner1}{black}{white}
\readableon{pbefsOnOwner2}{pbefsOwner2}{black}{white}
\readableon{pbefsOnOwner3}{pbefsOwner3}{black}{white}

\colorlet{pbBlue}{pbefsPrimary}
\colorlet{pbRed}{pbefsAdded}
\colorlet{pbYellow}{pbefsGoal}
