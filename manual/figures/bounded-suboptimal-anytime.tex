% bounded-suboptimal-anytime.tex: the paths of an anytime search over the
% expansions, the incumbent falling as a staircase and the lower bound
% rising until they meet at C*.
\documentclass[border=4pt]{standalone}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-pbefs}
\begin{document}
\begin{tikzpicture}[x=1cm, y=0.32cm]
  % the axes
  \draw[edge] (0,0) -- (10.4,0);
  \draw[edge] (0,0) -- (0,17.5);
  \node[val, anchor=north] at (10.2,-0.5) {expansions};
  \node[val, anchor=south] at (0,17.8) {cost};
  % C* and w C*
  \draw[pbefsInk!45, dashed, line width=1.1pt] (0,10) -- (10.2,10);
  \node[val, anchor=east] at (-0.15,10) {$C^*$};
  \draw[pbefsInk!45, dashed, line width=1.1pt] (0,16) -- (10.2,16);
  \node[val, anchor=east] at (-0.15,16) {$w C^*$};
  % the incumbent z, a staircase falling from the first path
  \draw[pbefsPrimary, line width=1.8pt]
    (1.2,14) -- (3.0,14) -- (3.0,12.5) -- (5.1,12.5) -- (5.1,11)
    -- (7.4,11) -- (7.4,10) -- (9.8,10);
  % the lower bound L, rising
  \draw[pbefsAdded, line width=1.8pt]
    (1.2,6) -- (3.0,6) -- (3.0,7.5) -- (5.1,7.5) -- (5.1,8.5)
    -- (7.4,8.5) -- (7.4,10) -- (9.8,10);
  % the paths found
  \foreach \x/\z/\l in {1.2/14/6, 3.0/12.5/7.5, 5.1/11/8.5, 7.4/10/10} {
    \fill[pbefsPrimary] (\x,\z) circle (2.6pt);
    \fill[pbefsAdded] (\x,\l) circle (2.6pt);
    }
  \node[val, anchor=south west, text=pbefsPrimary] at (1.25,14.2)
    {incumbent $z$};
  \node[val, anchor=north west, text=pbefsAdded] at (1.25,5.6)
    {lower bound $L \le C^*$};
  \node[val, anchor=west, align=left] at (7.55,12.2)
    {proven optimal:\\ $L = z = C^*$};
  \draw[decorate, decoration={brace, amplitude=4pt, mirror}]
    (3.25,7.5) -- node[val, right=5pt, fill=white] {$z / L$} (3.25,12.5);
  \node[val, anchor=north] at (1.2,-0.4) {first path};
\end{tikzpicture}
\end{document}
