% memory-bounded-paths.tex: the number of shortest paths from the corner to
% each cell of a 5 x 5 grid with 4 moves; a depth-first search without
% duplicate detection enters each cell once per path.
\documentclass[border=4pt]{standalone}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-pbefs}
\begin{document}
\begin{tikzpicture}
\foreach \j/\row in {0/{1,1,1,1,1}, 1/{1,2,3,4,5}, 2/{1,3,6,10,15},
                       3/{1,4,10,20,35}, 4/{1,5,15,35,70}}
 \foreach \n [count=\i from 0] in \row {
  \ifnum\n>9\relax
    \node[added, minimum width=8mm, minimum height=8mm] at (0.85*\i,-0.85*\j)
      {\n};
  \else
    \node[cell, minimum width=8mm, minimum height=8mm] at (0.85*\i,-0.85*\j)
      {\n};
  \fi
}
\node[touched, minimum width=8mm, minimum height=8mm] at (0,0) {$s_0$};
\node[val, anchor=west, align=left] at (4.0,-1.7)
  {cell $(i, j)$: $\binom{i+j}{i}$ shortest paths;\\
   the cells with at least 10 are in the rust color;\\
   one iteration of IDA* that reaches the far corner\\
   enters it 70 times along shortest paths alone};
\end{tikzpicture}
\end{document}
