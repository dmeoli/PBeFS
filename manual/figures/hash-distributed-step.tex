\begin{tikzpicture}

  \node[val, right=1pt of u] {$f = 5$};
  \node[unseen] (v1) at (1.1,-1.5) {$v_1$};

  \draw[edge] (u) -- (v2);
  \draw[edge] (u) -- (v3);
  \row{1.2}{-2.9}{OPEN$_0$}
