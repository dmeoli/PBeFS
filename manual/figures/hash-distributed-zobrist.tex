% \board{x}{y}{16 entries, 0 the blank}{cell of the moved tile}{cell left}
\newcommand{\board}[5]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \t [count=\c from 0] in {#3} {
      \pgfmathtruncatemacro{\r}{\c/4}
      \pgfmathtruncatemacro{\q}{mod(\c,4)}
      \ifnum\c=#4\relax
        \node[cellnew, minimum width=6mm, minimum height=6mm,
              font=\small] at (0.6*\q,-0.6*\r) {\t};
      \else\ifnum\c=#5\relax
        \node[cellnew, minimum width=6mm, minimum height=6mm,
              fill=pbefsAdded!40] at (0.6*\q,-0.6*\r) {};
      \else\ifnum\t=0\relax
        \node[cell, minimum width=6mm, minimum height=6mm]
          at (0.6*\q,-0.6*\r) {};
      \else
        \node[cell, minimum width=6mm, minimum height=6mm,
              font=\small] at (0.6*\q,-0.6*\r) {\t};
      \fi\fi\fi
    }
    \draw[pbefsPrimary, dashed, line width=1pt] (-0.4,-0.9) -- (2.2,-0.9);
  \end{scope}}
