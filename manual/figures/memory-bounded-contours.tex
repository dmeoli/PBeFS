\input{tikz-pbefs}

\fill[pbefsClosed] (1.2,0) ellipse (1.8cm and 1.1cm);
\node[current] (s) at (0.2,0) {$s_0$};
\node[goal] (t) at (6.2,1.18) {$t$};
\node[val, anchor=west] at (1.15,0.55) {$f \le \beta_1 = h(s_0)$};
\node[val, anchor=west] at (3.15,1.25) {$f \le \beta_2$};
\node[val, anchor=west] at (3.9,2.0) {$f \le \beta_3 = C^*$};
\draw[path] (s) .. controls (2.5,-0.9) and (4.8,-0.6) .. (t);
\node[val, anchor=north] at (2.6,-2.45)
  {iteration $k$ searches every path whose nodes have $f \le \beta_k$;
   $\beta_{k+1}$ is the least $f$ cut by iteration $k$};
