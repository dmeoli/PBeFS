\documentclass[border=4pt]{standalone}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-pbefs}
\begin{document}
\begin{tikzpicture}[s/.style={state, minimum size=4.2mm},
                    ln/.style={draw=pbefsInk!45, line width=1.1pt}]
  % the nblocks of 2 x 2 states, as boxes behind the states
  \foreach \a/\na in {0/0,1/2,2/4} \foreach \b/\nb in {0/0,1/2,2/4}
    \fill[pbefsFree, rounded corners=3pt]
      ($(0.9*\na, -0.9*\nb)+(-0.35,0.35)$) rectangle
      ($(0.9*\na+0.9, -0.9*\nb-0.9)+(0.35,-0.35)$);
  % the state space: a 6 x 6 grid of states
  \foreach \i in {0,...,5} \foreach \j [count=\k] in {0,...,4}
    \draw[ln] (0.9*\i, -0.9*\j) -- (0.9*\i, -0.9*\k);
  \foreach \j in {0,...,5} \foreach \i [count=\k] in {0,...,4}
    \draw[ln] (0.9*\i, -0.9*\j) -- (0.9*\k, -0.9*\j);
  \foreach \i in {0,...,5} \foreach \j in {0,...,5}
    \node[s] (s\i\j) at (0.9*\i, -0.9*\j) {};
  % one edge of the state space and its image
  \node[current, minimum size=4.2mm] at (s11) {};
  \node[reopened, minimum size=4.2mm] at (s21) {};
  \draw[path, shorten <=2.1mm, shorten >=2.1mm] (s11.center) --
    (s21.center);
  \node[caption, anchor=north] at (2.25,-5.0) {state space};
  % the abstract graph
  \begin{scope}[xshift=7.2cm, yshift=-0.9cm]
    \foreach \a/\l in {0/A,1/B,2/C} \foreach \b [count=\r] in {0,1,2}
      \node[node, minimum width=9mm, minimum height=9mm] (n\a\b) at
        (1.6*\a, -1.6*\b) {\l\r};
    \foreach \a in {0,1,2} {
      \draw[ln] (n\a0) -- (n\a1); \draw[ln] (n\a1) -- (n\a2);
      \draw[ln] (n0\a) -- (n1\a); \draw[ln] (n1\a) -- (n2\a);
    }
    \node[node-touched, minimum width=9mm, minimum height=9mm] at (n00)
      {A1};
    \node[node-added, minimum width=9mm, minimum height=9mm] at (n10)
      {B1};
    \draw[path] (n00) -- (n10);
    \node[caption, anchor=north] at (1.6,-3.95) {abstract graph};
  \end{scope}
  \draw[ptr] (5.0,-2.25) -- node[above=2pt, val]
    {$\mathrm{nblock}(\cdot)$} (6.4,-2.25);
\end{tikzpicture}
\end{document}
