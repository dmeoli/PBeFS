\documentclass[border=4pt]{standalone}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-pbefs}
\usetikzlibrary{decorations.pathreplacing}
\begin{document}
\begin{tikzpicture}[
    nb/.style={cell, minimum width=7mm, minimum height=7mm, inner sep=0pt,
               font=\sffamily\scriptsize},
    br/.style={decorate, decoration={brace, amplitude=4pt}, line width=0.8pt,
               draw=pbefsInk}]
  % \row{y}{b}{c}: the path 1..7, workers on b (owner 0) and c (owner 1)
  \newcommand{\row}[5]{
    \foreach \k in {1,...,7} {
      \node[nb] (#1n\k) at (1.2*\k, #2) {\k};
    }
    \foreach \k [count=\j from 2] in {1,...,6}
      \draw[draw=pbefsInk!45, line width=1.1pt] (#1n\k) -- (#1n\j);
    \node[ownerbox=0, minimum size=7mm, font=\sffamily\scriptsize]
      at (#1n#3) {#3};
    \node[ownerbox=1, minimum size=7mm, font=\sffamily\scriptsize]
      at (#1n#4) {#4};
    #5
  }
  % (a) 3 and 5 interfere: S(3) and S(5) share the nblock 4
  \row{a}{0}{3}{5}{
    \node[added, minimum width=7mm, minimum height=7mm, inner sep=0pt, font=\sffamily\scriptsize\bfseries] at (an4) {4};
    \draw[br] ($(an2.north west)+(0,0.12)$) -- node[above=4pt, val]
      {$S(3)$} ($(an4.north east)+(0,0.12)$);
    \draw[br] ($(an6.south east)+(0,-0.12)$) -- node[below=4pt, val]
      {$S(5)$} ($(an4.south west)+(0,-0.12)$);
  }
  \node[anchor=west, val, text width=4.6cm] at (9.0,0)
    {$S(3) \cap S(5) = \{4\}$: $5 \in I(3)$, and the two nblocks cannot be
     in use at the same time};
  % (b) 3 and 6 do not interfere
  \row{b}{-2.1}{3}{6}{
    \draw[br] ($(bn2.north west)+(0,0.12)$) -- node[above=4pt, val]
      {$S(3)$} ($(bn4.north east)+(0,0.12)$);
    \draw[br] ($(bn7.south east)+(0,-0.12)$) -- node[below=4pt, val]
      {$S(6)$} ($(bn5.south west)+(0,-0.12)$);
  }
  \node[anchor=west, val, text width=4.6cm] at (9.0,-2.1)
    {$S(3) \cap S(6) = \emptyset$: $6 \notin I(3)$, and two workers expand
     the nodes of 3 and of 6 with no lock};
  \node[steplabel, anchor=east] at (0.5,0) {(a)};
  \node[steplabel, anchor=east] at (0.5,-2.1) {(b)};
\end{tikzpicture}
\end{document}
